") | first | striptags | trim) if parts | length > 1 else "") -%} {%- if text | length > 160 -%} {{- (text[:157] | split(" ") | list)[:-1] | join(" ") -}}… {%- elif text | length > 40 -%} {{- text -}} {%- else -%} {{- config.site_description -}} {%- endif -%} {%- endif -%} {%- endmacro %} {#- Site-relative path (img/…) or full URL → full URL -#} {% macro absolute(path, config) -%} {{- path if "://" in path else config.site_url ~ path -}} {%- endmacro %}
